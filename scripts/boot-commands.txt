# boot-commands.txt — fed to the kernel's serial console by scripts/boot-gate.sh.
#
# One command per line, sent one at a time. Whatever they print becomes
# assertable in scripts/boot-milestones.txt, which is what makes the shell
# milestones mean "this command ran" rather than "a banner appeared".
uname -a
ps
ls /
ls /etc
cat /etc/motd
echo hello from home-os
#
# den, the shell. Each of these asserts on output that is not a substring of
# the command that produced it — the console echoes what is typed, so a
# milestone that also appears in the input would pass without the command
# having run at all. printf reassembles its arguments, which no echo of the
# input can produce.
printf "%s%s\n" quoted "  spaces"
GREETING=abc
printf "%s%s\n" $GREETING def
true && printf "%s%s\n" and ran
false || printf "%s%s\n" or ran
test 3 -lt 5 && printf "%s%s\n" cmp ok
[ -e /etc/motd ] && printf "%s%s\n" motd here
alias hi=printf
hi "%s%s\n" alias ok
printf "%s%s\n" red irect > /den.out
cat /den.out
cd /etc
printf "%s%s\n" cwd $PWD
cd /
run /bin/hello
exec /bin/hello_elf
exec /bin/echo hello from a Home program
exec /bin/cat /etc/motd
exec /bin/wc /etc/motd
exec /bin/seq 1 5
exec /bin/basename /usr/share/fonts/default.psf
exec /bin/dirname /usr/share/fonts/default.psf
exec /bin/nl /etc/motd
exec /bin/rev /etc/motd
exec /bin/head -n 1 /etc/motd
#
# tac, sort, uniq and cp, against a fixture with a deliberate out-of-order
# line and an adjacent duplicate (initramfs/etc/lines.txt). One-line input
# cannot tell these apart: it is the same output for all three.
exec /bin/tac /etc/lines.txt
exec /bin/sort /etc/lines.txt
exec /bin/uniq /etc/lines.txt
exec /bin/cp /etc/lines.txt /copied.txt
cat /copied.txt
#
# The rest of the coreutils. Each asserts on output that the command itself
# does not contain, and the expected values were taken from the host's own
# tools rather than from this implementation.
exec /bin/tail -n 2 /etc/lines.txt
exec /bin/cut -f 2 /etc/cols.txt
exec /bin/tr one xyz /etc/cols.txt
exec /bin/fold -w 4 /etc/motd
exec /bin/cmp /etc/motd /etc/sorted.txt
exec /bin/paste /etc/motd /etc/motd
exec /bin/sum /etc/motd
exec /bin/expand /etc/cols.txt
exec /bin/od /etc/motd
exec /bin/base64 /etc/motd
exec /bin/comm /etc/sorted.txt /etc/sorted2.txt
exec /bin/unexpand /etc/spaced.txt
exec /bin/strings /etc/motd
exec /bin/factor 84
exec /bin/printf %s-%s ab 7
exec /bin/expr 123 + 456
exec /bin/cksum /etc/motd
exec /bin/dd if=/etc/motd of=/dd.out bs=8 count=1
exec /bin/split -l 2 /etc/lines.txt /sp
cat /spaa
exec /bin/touch /touched.txt
/bin/test -e /touched.txt && printf "%s%s\n" touch ok
exec /bin/rm /touched.txt
/bin/test -e /touched.txt || printf "%s%s\n" rm ok
exec /bin/mkdir /newdir
ls /
exec /bin/rmdir /newdir
/bin/test 3 -gt 1 && printf "%s%s\n" test ok
/bin/test 1 -gt 3 || printf "%s%s\n" test neg
disk
ext2
ext2 hello.txt
mkfile written.txt kernel wrote this
remount
ext2 written.txt

hid
netc
