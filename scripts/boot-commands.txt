# boot-commands.txt — fed to the kernel's serial console by scripts/boot-gate.sh.
#
# One command per line, sent one at a time. Whatever they print becomes
# assertable in scripts/boot-milestones.txt, which is what makes the shell
# milestones mean "this command ran" rather than "a banner appeared".
uname
ps
ls /
ls /etc
cat /etc/motd
echo hello from home-os
run /bin/hello
exec /bin/hello_elf
exec /bin/echo hello from a Home program
exec /bin/cat /etc/motd
exec /bin/wc /etc/motd
exec /bin/seq 1 5
exec /bin/basename /usr/share/fonts/default.psf
exec /bin/dirname /usr/share/fonts/default.psf
exec /bin/nl /etc/motd
exec /bin/rev /etc/motd
exec /bin/head -n 1 /etc/motd
disk
ext2
ext2 hello.txt
mkfile written.txt kernel wrote this
remount
ext2 written.txt
irq
hid
netc
nets
