# boot-milestones.txt — the boot ratchet (scripts/boot-gate.sh).
#
# Substrings the kernel must print on the serial console, in this order, for
# the boot to count as unbroken. Each one is a subsystem that has been seen to
# initialize on real hardware emulation.
#
# Add milestones as the kernel reaches further. Never delete one to make CI
# green: a milestone that stops appearing means something that used to work
# has stopped working, which is exactly what this gate exists to catch.

# --- network RX path (STUB(S3) closure) ---------------------------------------
# netdev used to drop every frame it was handed, so arp, icmp, udp and tcp were
# all complete-looking and unreachable. These lines drive real frames through
# receive_packet() and check both that netdev demultiplexed them and that the
# protocol handler underneath accepted them. A regression to the old no-op
# drops a milestone rather than silently removing the RX path again.
[netdev] ARP demuxed, cache learned 10.0.2.15
[netdev] IPv4/ICMP demuxed and accepted by icmp
[netdev] IPv4/UDP demuxed and accepted by udp
[netdev] IPv4/TCP demuxed and accepted by tcp
[netdev] runt frame rejected, not dispatched
[netdev] bad IPv4 checksum rejected

# --- homefs on-disk structures (STUB(S4) step 1) ------------------------------
# docs/design/homefs.md §9 step 1 is "on-disk struct definitions + encode/decode
# unit tests". This is those tests run on the machine: a round trip through the
# byte layout, a torn write that must not read as valid, superblock slot
# selection, and a half-written commit record that must read as absent rather
# than as a transaction that happened.
[homefs] on-disk structures round-trip, 7/7

# --- xHCI controller discovery (STUB(S6) groundwork) --------------------------
# Found on the PCI bus by class, its register window mapped and its capability
# registers read. What remains of S6 is above this: the command ring, the event
# ring, the device-context base array, and enumeration. Those are sized from
# the numbers read here, which is why this part had to be right first.
[USB] controller registers readable

# --- userspace -------------------------------------------------------------
# `run /bin/hello` maps a flat binary into user pages, drops to ring 3, and
# the program's write and exit arrive through int $0x80. The exit code is 7
# so a real return can be told from a default one.
hello from ring 3
[User] exit(7)

# `exec` goes through the ELF loader instead: the image says where its
# segments live and the loader maps them there. The data-segment line proves
# filesz bytes were actually copied — a segment that arrives blank looks the
# same as one that was never read.
hello from an ELF at ring 3
elf data segment intact
[User] exit(9)


# `disk` writes a pattern to the last sector and reads it back. A driver that
# reports a device it cannot actually carry a byte to is the failure worth
# catching, and it looks identical at init time.
disk0: sector round-trip OK

# `ext2` walks a real ext2 filesystem on disk 0 — built by tools/mkext2.py,
# which also checks it afterwards, so the kernel is not grading its own work.
# The listing names entries that exist only on the disk, and the file's text
# comes off it.
11: hello.txt (type 1)
ext2 read path works

# The write path: a file created on disk, then the filesystem unmounted and
# mounted again — which discards the superblock, the group descriptors and
# every cached block — before the file is read back. Reading through the same
# memory that performed the write would prove only that memory remembers.
# tools/mkext2.py checks the image afterwards, including that the free-block
# count still matches the bitmap.
mkfile: created inode
remount: ok
kernel wrote this
