# boot-milestones.txt — the boot ratchet (scripts/boot-gate.sh).
#
# Substrings the kernel must print on the serial console, in this order, for
# the boot to count as unbroken. Each one is a subsystem that has been seen to
# initialize on real hardware emulation.
#
# Add milestones as the kernel reaches further. Never delete one to make CI
# green: a milestone that stops appearing means something that used to work
# has stopped working, which is exactly what this gate exists to catch.

# --- serial shell ----------------------------------------------------------
# Each line below is output from a command in scripts/boot-commands.txt, so it
# asserts that the shell parsed and ran that command over the serial line.
home-os x86_64
  PID  STATE  PRI

# The boot module is unpacked into the VFS, so `cat /etc/motd` reads bytes
# that came out of the cpio archive rather than anything compiled in.
# initramfs/etc/motd holds this line.
home-os initramfs works

hello from home-os

# --- userspace -------------------------------------------------------------
# `run /bin/hello` maps a flat binary into user pages, drops to ring 3, and
# the program's write and exit arrive through int $0x80. The exit code is 7
# so a real return can be told from a default one.
hello from ring 3
[User] exit(7)

# --- irq counters ----------------------------------------------------------
# `irq` reports which lines have actually fired. The keyboard one is only
# reachable because scripts/boot-gate.sh presses a key through QEMU's
# monitor — with -display none nothing else ever would.
[IRQ] timer line live
[IRQ] keyboard line live
